```ts
// Add to the imports near the top:
import { sql } from "drizzle-orm";
import { workflowPayments } from "@/lib/db/schema-payments";

// Add to imports (near top, with the others):
const VALID_SORTS = ["popular", "recent"] as const;
type CatalogSort = (typeof VALID_SORTS)[number] | undefined;

function readSort(value: string | null): CatalogSort {
  if (value === null) {
    return undefined;
  }
  const lower = value.toLowerCase();
  return (VALID_SORTS as readonly string[]).includes(lower)
    ? (lower as CatalogSort)
    : undefined;
}
```

For the ORDER BY swap, the popular path needs a GROUP BY join. Refactor the rows query to branch:

```ts
let rowsPromise: Promise<unknown[]>;
if (sort === "popular") {
  const callCountExpr = sql<number>`coalesce(count(${workflowPayments.id})::int, 0)`;
  rowsPromise = db
    .select({
      ...LISTED_WORKFLOW_COLUMNS,
      callCount: callCountExpr,
    })
    .from(workflows)
    .leftJoin(workflowPayments, eq(workflowPayments.workflowId, workflows.id))
    .where(whereClause)
    .groupBy(workflows.id)
    .orderBy(desc(callCountExpr), desc(workflows.id))
    .limit(limit)
    .offset(offset);
} else {
  // sort === "recent" OR sort === undefined (default behavior preserved)
  rowsPromise = db
    .select(LISTED_WORKFLOW_COLUMNS)
    .from(workflows)
    .where(whereClause)
    .orderBy(desc(workflows.listedAt))
    .limit(limit)
    .offset(offset);
}

const [countResult, rows] = await Promise.all([
  db.select({ count: count() }).from(workflows).where(whereClause),
  rowsPromise,
]);
```

The rest of the handler (sanitize description, build items array, NextResponse) stays unchanged. The `items` map will pass through the new optional `callCount` field when present (existing consumers ignore unknown fields per JSON tolerance).
