import { expect, test } from "@playwright/test";

test.use({ storageState: { cookies: [], origins: [] } });

test.describe("Tabbed Hub shell (HUBV2-02 / HUBV2-03 / HUBV2-08)", () => {
  test.beforeEach(async ({ context }) => {
    await context.clearCookies();
  });

  test("default tab on /hub is Protocols", async ({ page }) => {
    await page.goto("/hub", { waitUntil: "domcontentloaded" });
    const protocolsTab = page.getByRole("tab", { name: "Protocols" });
    await expect(protocolsTab).toHaveAttribute("data-state", "active", {
      timeout: 15_000,
    });
  });

  test("?tab=workflows lands on Workflows active", async ({ page }) => {
    await page.goto("/hub?tab=workflows", { waitUntil: "domcontentloaded" });
    const workflowsTab = page.getByRole("tab", { name: "Workflows" });
    await expect(workflowsTab).toHaveAttribute("data-state", "active", {
      timeout: 15_000,
    });
  });

  test("?tab=marketplace lands on Marketplace active", async ({ page }) => {
    await page.goto("/hub?tab=marketplace", { waitUntil: "domcontentloaded" });
    const marketplaceTab = page.getByRole("tab", { name: "Marketplace" });
    await expect(marketplaceTab).toHaveAttribute("data-state", "active", {
      timeout: 15_000,
    });
  });

  test("tab click updates URL via router.replace (history length stays constant)", async ({
    page,
  }) => {
    await page.goto("/hub", { waitUntil: "domcontentloaded" });
    const initialHistoryLength = await page.evaluate(() => history.length);

    await page.getByRole("tab", { name: "Workflows" }).click();
    await expect(page).toHaveURL(/[?&]tab=workflows/, { timeout: 5000 });

    await page.getByRole("tab", { name: "Marketplace" }).click();
    await expect(page).toHaveURL(/[?&]tab=marketplace/, { timeout: 5000 });

    const finalHistoryLength = await page.evaluate(() => history.length);
    // router.replace MUST NOT push new history entries.
    expect(finalHistoryLength).toBe(initialHistoryLength);
  });

  test("tab swap shows no skeleton flicker on the shell (HUBV2-02)", async ({
    page,
  }) => {
    await page.goto("/hub", { waitUntil: "domcontentloaded" });

    // Trigger a tab swap; assert no .animate-pulse element appears at the SHELL
    // level for >100ms. Tab CONTENT may have its own skeleton (e.g. Marketplace
    // initial fetch) — we only care that the SURROUNDING shell doesn't flicker.
    const pulseLocator = page.locator(
      'div:not([role="tabpanel"]) .animate-pulse'
    );

    await page.getByRole("tab", { name: "Workflows" }).click();
    // Wait briefly to let any hypothetical flicker render then check.
    await page.waitForTimeout(150);
    expect(await pulseLocator.count()).toBe(0);
  });

  test("HUBV2-08: no 'Listed in marketplace' badge anywhere in the Workflows tab", async ({
    page,
  }) => {
    await page.goto("/hub?tab=workflows", { waitUntil: "domcontentloaded" });
    // Wait for results to render (any tile or empty state).
    await page
      .locator('[data-view-mode]')
      .first()
      .waitFor({ state: "visible", timeout: 15_000 });
    const html = await page.content();
    expect(html).not.toMatch(/Listed in marketplace/i);
  });

  test("MARKETPLACE_BADGE_SLOT comment marker exists in source row", async () => {
    // Sanity check that the comment marker lives in the row component file —
    // a regression test against accidental deletion that future maintainers
    // might do without checking HUBV2-08.
    const fs = await import("node:fs/promises");
    const src = await fs.readFile(
      "components/hub/workflow-template-row.tsx",
      "utf8"
    );
    expect(src).toMatch(/MARKETPLACE_BADGE_SLOT/);
    expect(src).not.toMatch(/Listed in marketplace/);
  });
});
```

import { expect, test } from "@playwright/test";

test.use({ storageState: { cookies: [], origins: [] } });

const BANNED_TOKENS = [
  "creatorWalletAddress",
  "userId",
  "organizationId",
  "payerAddress",
  "amountUsdc",
] as const;

test.describe("Marketplace tab (MARKET-13)", () => {
  test.beforeEach(async ({ context }) => {
    await context.clearCookies();
  });

  test("loads /hub?tab=marketplace and renders header row", async ({ page }) => {
    await page.goto("/hub?tab=marketplace", {
      waitUntil: "domcontentloaded",
    });
    // Either a leaderboard renders, or the empty state. Both are acceptable.
    const leaderboard = page.getByRole("table", {
      name: "Marketplace leaderboard",
    });
    const empty = page.getByText("No paid services listed yet.");
    await expect(leaderboard.or(empty)).toBeVisible({ timeout: 15_000 });
  });

  test("page HTML does not leak sensitive columns (MARKET-04 / MARKET-13)", async ({
    page,
  }) => {
    await page.goto("/hub?tab=marketplace", {
      waitUntil: "domcontentloaded",
    });
    // Wait for marketplace render to settle.
    await page
      .getByRole("region", { name: "Marketplace results" })
      .waitFor({ state: "visible", timeout: 15_000 });
    const html = await page.content();
    for (const banned of BANNED_TOKENS) {
      expect(
        html,
        `Expected no leak of '${banned}' in /hub?tab=marketplace HTML`
      ).not.toContain(banned);
    }
  });

  test("sort dropdown switches between Popular and Newest", async ({
    page,
  }) => {
    await page.goto("/hub?tab=marketplace", {
      waitUntil: "domcontentloaded",
    });
    const trigger = page.getByRole("button", { name: "Sort marketplace by" });
    await expect(trigger).toBeVisible({ timeout: 15_000 });
    await expect(trigger).toContainText("Popular");

    await trigger.click();
    const newestOption = page.getByRole("menuitemradio", { name: "Newest" });
    await expect(newestOption).toBeVisible();
    await newestOption.click();
    await expect(page).toHaveURL(/sort=newest/, { timeout: 5000 });
    await expect(trigger).toContainText("Newest");

    await trigger.click();
    await page.getByRole("menuitemradio", { name: "Popular" }).click();
    await expect(page).toHaveURL(/sort=popular/, { timeout: 5000 });
    await expect(trigger).toContainText("Popular");
  });

  test("earnings sort is NOT in the dropdown (MARKET-02 deferred)", async ({
    page,
  }) => {
    await page.goto("/hub?tab=marketplace", {
      waitUntil: "domcontentloaded",
    });
    await page
      .getByRole("button", { name: "Sort marketplace by" })
      .click();
    // Assert there is no menu item containing "Earnings" / "earnings".
    const earningsOption = page.getByRole("menuitemradio", {
      name: /earnings/i,
    });
    await expect(earningsOption).toHaveCount(0);
  });

  test("tab swap to Workflows is instant (no shell skeleton flicker)", async ({
    page,
  }) => {
    await page.goto("/hub?tab=marketplace", {
      waitUntil: "domcontentloaded",
    });
    const pulseLocator = page.locator(
      'div:not([role="tabpanel"]) .animate-pulse'
    );

    await page.getByRole("tab", { name: "Workflows" }).click();
    await page.waitForTimeout(150);
    expect(await pulseLocator.count()).toBe(0);
    await expect(page).toHaveURL(/tab=workflows/);
  });
});
```
